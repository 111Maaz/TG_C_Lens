-- =============================================================================
-- Telangana Crime Lens - Minimal Working Schema (Aligned with Current Frontend)
-- =============================================================================
-- Notes:
-- - Matches current client behavior:
--   - Uses PostgreSQL POINT type for location, stored as "(lng,lat)" strings.
--   - Public read, insert, update, delete are allowed to work with the anon key
--     used by the frontend (no Supabase Auth gating in code right now).
-- - Includes useful indexes and updated_at trigger.
-- - Includes optional request_limits table (not used by current UI, but kept).
-- - Safe to run multiple times (IF NOT EXISTS / CREATE OR REPLACE used).
-- =============================================================================

-- Extensions required
CREATE EXTENSION IF NOT EXISTS pgcrypto;    -- for gen_random_uuid()

-- =============================================================================
-- Functions
-- =============================================================================

-- update_updated_at_column: keeps updated_at current on UPDATE
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at := NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Optional: simple rate limit helper (not used by current UI)
-- Keeps for parity with older migrations; you can remove if not needed.
CREATE OR REPLACE FUNCTION check_rate_limit(client_ip TEXT)
RETURNS BOOLEAN AS $$
DECLARE
  window_size   INTERVAL := INTERVAL '1 hour';
  max_requests  INTEGER  := 5;
  current_count INTEGER;
BEGIN
  DELETE FROM request_limits
  WHERE window_start < NOW() - window_size;

  INSERT INTO request_limits (ip_address, request_count, window_start)
  VALUES (client_ip, 1, NOW())
  ON CONFLICT (ip_address) DO UPDATE
    SET request_count =
      CASE
        WHEN request_limits.window_start < NOW() - window_size THEN 1
        ELSE request_limits.request_count + 1
      END,
        window_start =
      CASE
        WHEN request_limits.window_start < NOW() - window_size THEN NOW()
        ELSE request_limits.window_start
      END
  RETURNING request_count INTO current_count;

  RETURN current_count <= max_requests;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- =============================================================================
-- Tables
-- =============================================================================

-- 1) User-submitted crime reports
CREATE TABLE IF NOT EXISTS public.unofficial_reports (
  id             UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
  crime_category TEXT        NOT NULL,
  crime_type     TEXT        NOT NULL,
  district       TEXT        NOT NULL,
  description    TEXT        NOT NULL,
  -- POINT expects '(x,y)' strings; the app sends '(lng,lat)' and parses it back.
  location       POINT       NOT NULL,
  exact_location TEXT        NOT NULL,
  email          TEXT,
  is_anonymous   BOOLEAN     NOT NULL DEFAULT TRUE,
  status         TEXT        NOT NULL DEFAULT 'approved' CHECK (status IN ('pending','approved','rejected')),
  created_at     TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at     TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Trigger for updated_at
DROP TRIGGER IF EXISTS update_unofficial_reports_updated_at ON public.unofficial_reports;
CREATE TRIGGER update_unofficial_reports_updated_at
BEFORE UPDATE ON public.unofficial_reports
FOR EACH ROW
EXECUTE FUNCTION update_updated_at_column();

-- Useful indexes for common queries
CREATE INDEX IF NOT EXISTS idx_unofficial_reports_created_at ON public.unofficial_reports (created_at DESC);
CREATE INDEX IF NOT EXISTS idx_unofficial_reports_status      ON public.unofficial_reports (status);
CREATE INDEX IF NOT EXISTS idx_unofficial_reports_district    ON public.unofficial_reports (district);

-- 2) Optional rate-limiting table (not used by current UI)
CREATE TABLE IF NOT EXISTS public.request_limits (
  ip_address   TEXT        PRIMARY KEY,
  request_count INTEGER     NOT NULL DEFAULT 1,
  window_start TIMESTAMPTZ  NOT NULL DEFAULT NOW()
);

-- =============================================================================
-- Row Level Security (RLS) and Policies
-- =============================================================================
-- The current app uses the anon key and does not sign in with Supabase Auth.
-- To make the app work out-of-the-box:
-- - Allow PUBLIC (anon) SELECT/INSERT/UPDATE/DELETE on unofficial_reports.
-- If you later add auth and admin roles, lock these down accordingly.

ALTER TABLE public.unofficial_reports ENABLE ROW LEVEL SECURITY;

DO $$
BEGIN
  -- Allow everyone to read reports
  IF NOT EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'public' AND tablename = 'unofficial_reports' AND policyname = 'unofficial_reports_select_all'
  ) THEN
    CREATE POLICY unofficial_reports_select_all
      ON public.unofficial_reports
      FOR SELECT
      USING (true);
  END IF;

  -- Allow everyone to insert reports
  IF NOT EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'public' AND tablename = 'unofficial_reports' AND policyname = 'unofficial_reports_insert_all'
  ) THEN
    CREATE POLICY unofficial_reports_insert_all
      ON public.unofficial_reports
      FOR INSERT
      WITH CHECK (true);
  END IF;

  -- Allow everyone to update (needed for status moderation from the UI)
  IF NOT EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'public' AND tablename = 'unofficial_reports' AND policyname = 'unofficial_reports_update_all'
  ) THEN
    CREATE POLICY unofficial_reports_update_all
      ON public.unofficial_reports
      FOR UPDATE
      USING (true)
      WITH CHECK (true);
  END IF;

  -- Allow everyone to delete (admin UI deletes from the client)
  IF NOT EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'public' AND tablename = 'unofficial_reports' AND policyname = 'unofficial_reports_delete_all'
  ) THEN
    CREATE POLICY unofficial_reports_delete_all
      ON public.unofficial_reports
      FOR DELETE
      USING (true);
  END IF;
END
$$;

-- =============================================================================
-- Realtime (Supabase)
-- =============================================================================
-- Most Supabase projects already have publication "supabase_realtime".
-- Ensure the table is included.
DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM pg_catalog.pg_publication WHERE pubname = 'supabase_realtime') THEN
    ALTER PUBLICATION supabase_realtime ADD TABLE public.unofficial_reports;
  END IF;
END
$$;

-- =============================================================================
-- Idempotent column reconciliation (in case you’re restoring into an existing DB)
-- =============================================================================
-- This keeps the table aligned if it existed with fewer columns.
DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema='public' AND table_name='unofficial_reports' AND column_name='crime_category') THEN
    ALTER TABLE public.unofficial_reports ADD COLUMN crime_category TEXT NOT NULL DEFAULT 'Unknown';
  END IF;

  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema='public' AND table_name='unofficial_reports' AND column_name='exact_location') THEN
    ALTER TABLE public.unofficial_reports ADD COLUMN exact_location TEXT NOT NULL DEFAULT 'Not specified';
  END IF;

  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema='public' AND table_name='unofficial_reports' AND column_name='email') THEN
    ALTER TABLE public.unofficial_reports ADD COLUMN email TEXT;
  END IF;

  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema='public' AND table_name='unofficial_reports' AND column_name='status') THEN
    ALTER TABLE public.unofficial_reports ADD COLUMN status TEXT NOT NULL DEFAULT 'approved' CHECK (status IN ('pending','approved','rejected'));
  END IF;

  IF NOT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema='public' AND table_name='unofficial_reports' AND column_name='updated_at') THEN
    ALTER TABLE public.unofficial_reports ADD COLUMN updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW();
  END IF;
END
$$;
